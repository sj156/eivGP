Creating the eivGP R package

2 Package setup

This chapter owns package metadata and clears only generated package directories before the audited modules are recopied. The source scripts under script/ and the standalone analysis documents are never removed.

usethis::create_package(
  path = ".",
  fields = list(
    Package = params$package_name,
    Version = "0.1.0",
    Title = "Errors-in-Variables Gaussian Processes for Ordinal Mixed Inputs",
    Description = paste(
      "Fits errors-in-variables Gaussian process regressions with numeric",
      "predictors, ordinal proxies for latent continuous inputs, and sparse",
      "calibration measurements. Includes reproducible Study I and Study II",
      "simulation workflows and audited mixed-input GP competitors."
    ),
    `Authors@R` = c(
      person(given = "Penghui", family = "Fu", role = "aut"),
      person(given = "Xinye", family = "Chen", role = "aut"),
      person(given = "Jianhua Z.", family = "Huang", role = "aut"),
      person(
        given = "Sheng", family = "Jiang",
        email = "jiangsheng@cuhk.edu.cn", role = c("aut", "cre")
      )
    ),
    License = "MIT + file LICENSE",
    Encoding = "UTF-8",
    Depends = "R (>= 4.1.0)",
    Imports = paste(
      "methods, parallel, posterior, stats, tools, TruncatedNormal, utils"
    ),
    Suggests = paste(
      "dplyr, EzGP, ggplot2, kergp, knitr, LVGP, patchwork, pkgload,",
      "rmarkdown, testthat, tidyr"
    ),
    `Config/testthat/edition` = "3"
  )
)

package_dir <- usethis::proj_get()
writeLines(
  c("^.*[.]Rproj$", "^[.]Rproj[.]user$"),
  file.path(package_dir, ".Rbuildignore")
)
generated_directories <- c(
  file.path(package_dir, "R"),
  file.path(package_dir, "man"),
  file.path(package_dir, "tests")
)
for (directory in generated_directories) {
  if (dir.exists(directory)) unlink(directory, recursive = TRUE)
}
generated_resources <- c(
  file.path(package_dir, "inst", "experiments"),
  file.path(package_dir, "inst", "simulation-source")
)
for (directory in generated_resources) {
  if (dir.exists(directory)) unlink(directory, recursive = TRUE)
}
dir.create(file.path(package_dir, "R"), recursive = TRUE, showWarnings = FALSE)

writeLines(
  c(
    "YEAR: 2026",
    paste(
      "COPYRIGHT HOLDER: Penghui Fu, Xinye Chen, Jianhua Z. Huang,",
      "and Sheng Jiang"
    )
  ),
  file.path(package_dir, "LICENSE")
)
repository_dir <- dirname(knitr::current_input(dir = TRUE))
stopifnot(file.copy(
  file.path(repository_dir, "LICENSE"),
  file.path(package_dir, "LICENSE.md"),
  overwrite = TRUE
))

writeLines(
  c(
    "# eivGP",
    "",
    "`eivGP` implements errors-in-variables Gaussian process regression",
    "with ordinal proxies for latent continuous inputs and sparse calibration.",
    "",
    "## Stable model interface",
    "",
    "```r",
    "fit <- fit_eivgp(X, y, C, U_obs, engine = 'multivariate')",
    "mean_draws <- predict(fit, new_X = X_new, new_C = C_new, target = 'mean')",
    "response_draws <- predict(fit, new_X = X_new, new_C = C_new, target = 'response')",
    "latent_draws <- impute_eivgp(fit, new_C = C_new)",
    "```",
    "",
    "## Publication workflows",
    "",
    "Both exported runners default to a read-only dry run:",
    "",
    "```r",
    "run_study1_simulation()",
    "run_study2_simulation()",
    "```",
    "",
    "Use `study1_simulation_config()` or `study2_simulation_config()` to",
    "select data generation, smoke testing, or the frozen publication design.",
    "Smoke and publication fits automatically run the exact competitor and",
    "paired-design validators before computation; publication requires every",
    "competitor validation to succeed. Results are archived under `config/`.",
    "See the repository's standalone experiment R Markdown files for the full",
    "top-to-bottom reproduction workflow."
  ),
  file.path(package_dir, "README.md")
)

writeLines(
  c(
    "# eivGP 0.1.0",
    "",
    "- Added stable fitting, prediction, imputation, and competitor interfaces.",
    "- Added installed-package Study I and Study II publication runners.",
    "- Added frozen-data checksums, source fingerprints, resumable outputs,",
    "  paired comparisons, and strict publication diagnostic gates.",
    "- Added automatic, archived competitor and design validation before all",
    "  smoke and publication fits.",
    "- Added a package-facing real-data interface and standalone application",
    "  skeleton."
  ),
  file.path(package_dir, "NEWS.md")
)

dir.create(file.path(package_dir, "inst"), recursive = TRUE, showWarnings = FALSE)
writeLines(

  ),
  file.path(package_dir, "inst", "CITATION")
)
